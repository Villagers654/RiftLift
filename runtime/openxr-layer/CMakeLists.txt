cmake_minimum_required(VERSION 3.20)
project(RiftLiftOpenXRLayer LANGUAGES CXX)
find_path(OPENXR_INCLUDE_DIR openxr/openxr_loader_negotiation.h REQUIRED)
add_library(RiftLiftOpenXRLayer SHARED OpenXRLayer.cpp)
target_include_directories(RiftLiftOpenXRLayer PRIVATE "${OPENXR_INCLUDE_DIR}")
target_compile_features(RiftLiftOpenXRLayer PRIVATE cxx_std_17)
set_target_properties(RiftLiftOpenXRLayer PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MINGW)
  target_link_options(RiftLiftOpenXRLayer PRIVATE -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(OpenXRLayerTests LayerTests.cpp)
  target_include_directories(OpenXRLayerTests PRIVATE "${OPENXR_INCLUDE_DIR}")
  target_compile_features(OpenXRLayerTests PRIVATE cxx_std_17)
  target_link_libraries(OpenXRLayerTests PRIVATE RiftLiftOpenXRLayer)
  set_target_properties(OpenXRLayerTests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(OpenXRLayerTests PRIVATE -static)
  endif()
  add_test(NAME OpenXRLayerIsolation COMMAND OpenXRLayerTests)
endif()
